function doPost(e) {
  try {
    // リクエストボディからJSONデータを取得 (x-www-form-urlencoded と raw JSON の両方に対応)
    var jsonString = e.parameter.data || (e.postData ? e.postData.contents : "");
    var rows = JSON.parse(jsonString);
    
    // スプレッドシートを取得
    // ※「TypeError: Cannot read properties of null (reading 'getActiveSheet')」というエラーが出る場合、
    // スクリプトがスプレッドシートと紐付いていません。
    // その場合は以下の getActiveSpreadsheet() の行をコメントアウトし、
    // openById() の行のコメントを外して、スプレッドシートのIDを指定してください。
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
    // var sheet = SpreadsheetApp.openById("ここにスプレッドシートのIDを貼り付け").getSheets()[0];
    
    // rowsは二次元配列を想定：[[通し番号, 決済時刻, 商品1~5, 合計個数, 合計金額], ...]
    if (Array.isArray(rows) && rows.length > 0) {
      var startRow = sheet.getLastRow() + 1;
      var numRows = rows.length;
      var numCols = rows[0].length;
      
      // まとめて書き込み（パフォーマンス向上のため）
      sheet.getRange(startRow, 1, numRows, numCols).setValues(rows);
    }
    
    // 成功レスポンスを返す
    return ContentService.createTextOutput(JSON.stringify({"status": "success"}))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    // エラーレスポンスを返す
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.message}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
